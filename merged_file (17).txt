getExactPingEjeT = function()
    if #KinematicBuffer.pings == 0 then return 0.035 end
    local sum = 0
    for _, v in ipairs(KinematicBuffer.pings) do sum = sum + v end
    return sum / #KinematicBuffer.pings
end

isPlayerAlive = function(p)
    return p and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 and p.Character:FindFirstChild("HumanoidRootPart") ~= nil
end

checkPlayerTool = function(player, toolName)
    if not player then return false end
    local bp = player:FindFirstChild("Backpack")
    local char = player.Character

    if bp then
        if bp:FindFirstChild(toolName) then return true end
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") and (t.Name == toolName or string.find(string.lower(t.Name), string.lower(toolName))) then
                return true
            end
        end
    end

    if char then
        if char:FindFirstChild(toolName) then return true end
        for _, t in ipairs(char:GetChildren()) do
            if t:IsA("Tool") and (t.Name == toolName or string.find(string.lower(t.Name), string.lower(toolName))) then
                return true
            end
        end
    end

    return false
end

isPlayerInLobby = function()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return true end
    local map = getMap()
    if not map then return true end

    local coinCont = map:FindFirstChild("CoinContainer")
    local spawns = map:FindFirstChild("Spawns")
    if not coinCont and not spawns then return true end

    if spawns and #spawns:GetChildren() > 0 then
        local closestDist = math.huge
        for _, sp in ipairs(spawns:GetChildren()) do
            if sp:IsA("BasePart") then
                local dist = (hrp.Position - sp.Position).Magnitude
                if dist < closestDist then closestDist = dist end
            end
        end
        if closestDist > 550 then return true end
    end

    if hrp.Position.Y > 120 and (hrp.Position - Vector3.new(-107, 152, 41)).Magnitude < 140 then
        return true
    end

    return false
end

findNearestPlayer = function()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    if not isPlayerAlive(LocalPlayer) then return nil end
    local myPos = LocalPlayer.Character.HumanoidRootPart.Position

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isPlayerAlive(player) then
            local otherRoot = player.Character.HumanoidRootPart
            local distance = (myPos - otherRoot.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                nearestPlayer = player
            end
        end
    end
    return nearestPlayer
end

checkUniversalLoS = function(origin, targetPos, targetChar)
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, targetChar}
    rayParams.IgnoreWater = true

    local hit = Workspace:Raycast(origin, targetPos - origin, rayParams)
    if hit and hit.Instance and (hit.Instance.CanCollide == false or hit.Instance.Transparency > 0.7) then
        return true
    end
    return hit == nil
end

getExtreme4AxisTarget = function(targetChar, origin)
    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return nil end

    local targetBone = nil
    local bonesToScan = {"HumanoidRootPart", "UpperTorso", "Head", "LowerTorso"}

    if consecutiveMisses > 0 or hum.FloorMaterial == Enum.Material.Air then
        bonesToScan = {
            "HumanoidRootPart", "UpperTorso", "Head", "LowerTorso",
            "RightUpperArm", "LeftUpperArm", "RightLowerLeg", "LeftLowerLeg", "RightFoot", "LeftFoot"
        }
    end

    for _, boneName in ipairs(bonesToScan) do
        local part = targetChar:FindFirstChild(boneName)
        if part and (quantumWallPierce or consecutiveMisses > 0 or checkUniversalLoS(origin, part.Position, targetChar)) then
            targetBone = part
            break
        end
    end

    if not targetBone then
        targetBone = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("UpperTorso")
        if not targetBone then return nil end
    end

    local basePos = targetBone.Position
    local hrp = targetChar:FindFirstChild("HumanoidRootPart") or targetBone
    local dist = (basePos - origin).Magnitude

    local player = Players:GetPlayerFromCharacter(targetChar)
    local posBuf = player and KinematicBuffer.positions[player]

    local vX, vY, vZ = 0, 0, 0
    local aX, aY, aZ = 0, 0, 0

    if posBuf and #posBuf >= 8 then
        local pNew = posBuf[#posBuf]
        local pMid = posBuf[#posBuf - 4]
        local pOld = posBuf[#posBuf - 7]

        local dT1 = pNew.t - pMid.t
        local dT2 = pMid.t - pOld.t

        if dT1 > 0 and dT2 > 0 then
            local vX1 = (pNew.pos.X - pMid.pos.X) / dT1
            local vY1 = (pNew.pos.Y - pMid.pos.Y) / dT1
            local vZ1 = (pNew.pos.Z - pMid.pos.Z) / dT1

            local vX2 = (pMid.pos.X - pOld.pos.X) / dT2
            local vY2 = (pMid.pos.Y - pOld.pos.Y) / dT2
            local vZ2 = (pMid.pos.Z - pOld.pos.Z) / dT2

            vX, vY, vZ = vX1, vY1, vZ1

            local totalDT = (pNew.t - pOld.t) * 0.5
            aX = (vX1 - vX2) / totalDT
            aY = (vY1 - vY2) / totalDT
            aZ = (vZ1 - vZ2) / totalDT

            -- [[ FILTRO ANTI-ZIGZAG COMPACTO ]]
            if not KinematicBuffer.accelerations[player] then
                KinematicBuffer.accelerations[player] = {}
            end
            local accBuf = KinematicBuffer.accelerations[player]
            table.insert(accBuf, Vector3.new(aX, aY, aZ))
            if #accBuf > 5 then table.remove(accBuf, 1) end

            if #accBuf >= 3 then
                local lastAcc = accBuf[#accBuf]
                local prevAcc = accBuf[#accBuf - 1]
                local isZigZag = (lastAcc.X * prevAcc.X < 0) or (lastAcc.Z * prevAcc.Z < 0)

                if isZigZag then
                    local sumAcc = Vector3.zero
                    for _, accVal in ipairs(accBuf) do sumAcc = sumAcc + accVal end
                    local avgAcc = sumAcc / #accBuf
                    aX = avgAcc.X * 0.4
                    aY = avgAcc.Y * 0.6
                    aZ = avgAcc.Z * 0.4
                end
            end
        end
    else
        local rawVel = hrp.AssemblyLinearVelocity
        vX, vY, vZ = rawVel.X, rawVel.Y, rawVel.Z
    end

    local pingEjeT = getExactPingEjeT()
    local adaptativeRatio = (consecutiveMisses > 0) and 1.30 or 1.10
    local flightTime = (dist / BULLET_SPEED) + (pingEjeT * adaptativeRatio)

    if math.abs(vY) > 4 or hum.MoveDirection.Magnitude > 0.1 then
        aX = aX * 1.25
        aZ = aZ * 1.25
        aY = aY * 1.40
    end

    local leadX = (vX * flightTime) + (0.5 * aX * (flightTime ^ 2))
    local leadZ = (vZ * flightTime) + (0.5 * aZ * (flightTime ^ 2))
    local leadY = 0

    local inAir = (hum.FloorMaterial == Enum.Material.Air) or (math.abs(vY) > 1.0)
    if inAir then
        local g = Workspace.Gravity > 0 and Workspace.Gravity or 196.2
        leadY = (vY * flightTime) + (0.5 * (aY - g) * (flightTime ^ 2))
    else
        leadY = (vY * flightTime) + (0.5 * aY * (flightTime ^ 2))
    end

    local finalLeadVector = Vector3.new(leadX, leadY, leadZ)
    local maxMove = (hum.WalkSpeed > 0 and hum.WalkSpeed or 16) * flightTime * 1.5
    if finalLeadVector.Magnitude > maxMove then
        finalLeadVector = finalLeadVector.Unit * maxMove
    end

    return basePos + finalLeadVector
end

runExtremeShootAction = function(silent)
    if shootActionDebounce then return false end
    shootActionDebounce = true

    local char = LocalPlayer.Character
    if not char then shootActionDebounce = false return false end

    local hum = char:FindFirstChildOfClass("Humanoid")
    local myHRP = char:FindFirstChild("HumanoidRootPart")
    if not hum or not myHRP or hum.Health <= 0 then
        shootActionDebounce = false
        return false
    end

    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local gun = char:FindFirstChild("Gun") or (backpack and backpack:FindFirstChild("Gun"))

    if not gun then
        if not silent then Notify("Shoot", "No tienes el arma en inventario.", 2) end
        shootActionDebounce = false
        return false
    end

    if not gun.ManualActivationOnly then
        pcall(function() gun.ManualActivationOnly = true end)
    end

    if gun.Parent == backpack then
        hum:EquipTool(gun)
        task.wait(0.04)
    end

    local murderer = cachedMurderer or findMurderer()
    if not murderer or not murderer.Character or not isPlayerAlive(murderer) then
        if not silent then Notify("Shoot", "Murderer no detectado en ronda.", 2) end
        shootActionDebounce = false
        return false
    end

    local mHRP = murderer.Character:FindFirstChild("HumanoidRootPart") or murderer.Character:FindFirstChild("UpperTorso")
    if not mHRP then
        shootActionDebounce = false
        return false
    end

    local realOrigin = myHRP.Position + (myHRP.CFrame.LookVector * 1.2)
    local targetPos = getExtreme4AxisTarget(murderer.Character, realOrigin)

    if not targetPos then
        targetPos = mHRP.Position
        consecutiveMisses = consecutiveMisses + 1
    else
        consecutiveMisses = 0
    end

    local hasLoS = checkUniversalLoS(realOrigin, targetPos, murderer.Character)
    local finalOrigin = realOrigin

    if not hasLoS then
        if quantumWallPierce then
            finalOrigin = targetPos - (mHRP.CFrame.LookVector * 1.5)
        else
            if not silent then
                Notify("Apex Shoot", "Murderer cubierto por pared. Tiro protegido.", 2.5)
            end
            shootActionDebounce = false
            return false
        end
    end

    local aimLookCFrame = CFrame.lookAt(finalOrigin, targetPos)
    local targetCFrame = CFrame.new(targetPos)

    local shootRemote = gun:FindFirstChild("Shoot")
    if shootRemote then

runAimbotShiftLock = function(enable)
    aimbotEnabled = enable
    if aimbotConn then aimbotConn:Disconnect() aimbotConn = nil end
    if not enable then
        pcall(function() UserInputService.MouseBehavior = Enum.MouseBehavior.Default end)
        return
    end

    aimbotConn = RunService.RenderStepped:Connect(function(deltaTime)
        if not aimbotEnabled or not scriptActive then return end
        local murderer = cachedMurderer or findMurderer()
        if not murderer or not murderer.Character or not isPlayerAlive(murderer) then return end

        local mChar = murderer.Character
        local mTorso = mChar:FindFirstChild("UpperTorso") or mChar:FindFirstChild("Torso") or mChar:FindFirstChild("HumanoidRootPart")
        if not mTorso then return end

        local myChar = LocalPlayer.Character
        local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
        local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
        if not myHRP or not myHum or myHum.Health <= 0 then return end

        pcall(function()
            UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
            local lookPos = Vector3.new(mTorso.Position.X, myHRP.Position.Y, mTorso.Position.Z)
            local targetCharCF = CFrame.lookAt(myHRP.Position, lookPos)
            myHRP.CFrame = myHRP.CFrame:Lerp(targetCharCF, math.clamp(deltaTime * 18, 0, 1))

            local camPos = Camera.CFrame.Position
            local aimTargetCF = CFrame.lookAt(camPos, mTorso.Position)
            Camera.CFrame = Camera.CFrame:Lerp(aimTargetCF, math.clamp(deltaTime * 24, 0, 1))
        end)
    end)
end

setGhostTransparency = function(char, transValue)
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            if transValue == 0.5 then
                if originalTransparencies[part] == nil then originalTransparencies[part] = part.Transparency end
                part.Transparency = 0.5
            else
                part.Transparency = originalTransparencies[part] or 0
            end
        elseif part:IsA("Decal") then
            if transValue == 0.5 then
                if originalTransparencies[part] == nil then originalTransparencies[part] = part.Transparency end
                part.Transparency = 0.5
            else
                part.Transparency = originalTransparencies[part] or 0
            end
        end
    end
    if transValue == 0 then table.clear(originalTransparencies) end
end

applyHighlightToCharacter = function(player, character)
    if not espActive or not character or player == LocalPlayer then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")

    if not humanoid or humanoid.Health <= 0 then
        if ESPInternal.billboards[character] then ESPInternal.billboards[character]:Destroy() ESPInternal.billboards[character] = nil end
        if ESPInternal.highlights[character] then ESPInternal.highlights[character]:Destroy() ESPInternal.highlights[character] = nil end
        return
    end

    local hasKnife = checkPlayerTool(player, "Knife")
    local hasGun = checkPlayerTool(player, "Gun")
    local isMurderer = hasKnife or (cachedMurderer == player) or (findMurderer() == player)
    local isSheriff = false
    local isHero = false

    if not isMurderer then
        if hasGun then
            local originalRole = (playerDataESP[player.Name] and (playerDataESP[player.Name].Role or playerDataESP[player.Name].role))
                or (gamePlayerData[player.Name] and (gamePlayerData[player.Name].Role or gamePlayerData[player.Name].role))
            if originalRole and originalRole ~= "Sheriff" then isHero = true else isSheriff = true end
        else
            local sh = findSheriff()
            if sh == player then isSheriff = true end
        end
    end

    if ignoreInnocents and not isMurderer and not isSheriff and not isHero then
        if ESPInternal.highlights[character] then ESPInternal.highlights[character]:Destroy() ESPInternal.highlights[character] = nil end
        if ESPInternal.billboards[character] then ESPInternal.billboards[character]:Destroy() ESPInternal.billboards[character] = nil end
        return
    end

    local hl = ESPInternal.highlights[character]
    if not hl or not hl.Parent then
        if hl then hl:Destroy() end
        hl = Instance.new("Highlight")
        hl.Name = "Player_Highlight"
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 1
        hl.OutlineTransparency = 0
        hl.Adornee = character
        hl.Parent = PARENT
        ESPInternal.highlights[character] = hl
    end

    local targetColor = Color3.fromRGB(0, 255, 8)
    local roleText = nil
    local textColor = Color3.fromRGB(255, 255, 255)

    if isMurderer then
        targetColor = Color3.fromRGB(255, 40, 40)
        roleText = "MURDERER"
        textColor = Color3.fromRGB(255, 85, 85)
    elseif isSheriff then
        targetColor = Color3.fromRGB(40, 140, 255)
        roleText = "SHERIFF"
        textColor = Color3.fromRGB(80, 185, 255)
    elseif isHero then
        targetColor = Color3.fromRGB(0, 210, 255)
        roleText = "HERO"
        textColor = Color3.fromRGB(95, 220, 255)
    end

    if hl.OutlineColor ~= targetColor then hl.OutlineColor = targetColor end
    hl.FillTransparency = 1
    hl.OutlineTransparency = 0
    hl.Enabled = true

    if roleText and head then
        local bg = ESPInternal.billboards[character]
        if not bg or not bg.Parent then
            if bg then bg:Destroy() end
            bg = Instance.new("BillboardGui")
            bg.Name = "Role_Billboard"
            bg.AlwaysOnTop = true
            bg.Size = UDim2.new(0, 140, 0, 30)
            bg.StudsOffset = Vector3.new(0, 2.5, 0)
            bg.Adornee = head
            bg.Parent = PARENT

            local lbl = Instance.new("TextLabel")
            lbl.Name = "RoleLabel"
            lbl.Size = UDim2.new(1, 0, 1, 0)
            lbl.BackgroundTransparency = 1
            lbl.Font = Enum.Font.GothamBold
            lbl.TextSize = 13
            lbl.Parent = bg

            local stroke = Instance.new("UIStroke")
            stroke.Thickness = 1.5
            stroke.Color = Color3.fromRGB(0, 0, 0)
            stroke.Parent = lbl
            ESPInternal.billboards[character] = bg
        end
        local lbl = bg:FindFirstChild("RoleLabel")
        if lbl then
            lbl.Text = roleText
            lbl.TextColor3 = textColor
        end
    else
        if ESPInternal.billboards[character] then
            ESPInternal.billboards[character]:Destroy()
            ESPInternal.billboards[character] = nil
        end
    end
end

            playerDataESP = data
            reloadESP()
        end)
    end

    local function setupPlayerESP(player)
        if player == LocalPlayer then return end
        if player.Character then task.spawn(applyHighlightToCharacter, player, player.Character) end
        local pConn = player.CharacterAdded:Connect(function(character)
            task.wait(0.04)
            if espActive then applyHighlightToCharacter(player, character) end
        end)
        if not playerConnections[player] then playerConnections[player] = {} end
        table.insert(playerConnections[player], pConn)
    end

    playerAddedConnectionESP = Players.PlayerAdded:Connect(setupPlayerESP)
    for _, player in ipairs(Players:GetPlayers()) do setupPlayerESP(player) end

    espLoop = task.spawn(function()
        while espActive and scriptActive do
            reloadESP()
            task.wait(0.20)
        end
    end)
    reloadESP()
end

stopESP = function()
    espActive = false
    if espLoop then task.cancel(espLoop) espLoop = nil end
    clearESP()
    if databaseConnectionESP then databaseConnectionESP:Disconnect() databaseConnectionESP = nil end
    if playerAddedConnectionESP then playerAddedConnectionESP:Disconnect() playerAddedConnectionESP = nil end
end

executeFlingAction = function(TargetPlayer, customReturnCFrame)
    if not TargetPlayer or TargetPlayer == LocalPlayer or flingInProgress then return end
    if not isPlayerAlive(LocalPlayer) or not isPlayerAlive(TargetPlayer) then return end

    flingInProgress = true
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart

    local TCharacter = TargetPlayer.Character
    local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    local THead = TCharacter and TCharacter:FindFirstChild("Head")

    if not Character or not Humanoid or not RootPart or not TCharacter or not THumanoid then
        flingInProgress = false
        return
    end

    local oldPos = customReturnCFrame or RootPart.CFrame
    local oldFPDH = Workspace.FallenPartsDestroyHeight
    local Cam = Workspace.CurrentCamera

    pcall(function()
        if RootPart.Velocity.Magnitude < 50 then getgenv().OldPos = oldPos end
        if THead then Cam.CameraSubject = THead elseif THumanoid then Cam.CameraSubject = THumanoid end
        if not TCharacter:FindFirstChildWhichIsA("BasePart") then return end

        local function FPos(BasePart, Pos, Ang)
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
            RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
            RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        end

        local function SFBasePart(BasePart)
            local TimeToWait = 2.5
            local Time = tick()
            local Angle = 0

            repeat
                if RootPart and THumanoid then
                    Angle = Angle + 100
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                else
                    break
                end
            until BasePart.Velocity.Magnitude > 500 or BasePart.Position.Y < -40 or BasePart.Parent ~= TargetPlayer.Character or TargetPlayer.Parent ~= Players or TargetPlayer.Character ~= TCharacter or THumanoid.Sit or Humanoid.Health <= 0 or tick() > Time + TimeToWait
        end

        Workspace.FallenPartsDestroyHeight = 0/0
        local BV = Instance.new("BodyVelocity")
        BV.Name = "EpixVel"
        BV.Parent = RootPart
        BV.Velocity = Vector3.new(9e8, 9e8, 9e8)
        BV.MaxForce = Vector3.new(1/0, 1/0, 1/0)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        if TRootPart then SFBasePart(TRootPart) elseif THead then SFBasePart(THead) end

        BV:Destroy()
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
        Cam.CameraSubject = Humanoid

        local returnTimeout = tick() + 0.40
        repeat
            if RootPart and Character then
                local fallbackPos = oldPos * CFrame.new(0, .5, 0)
                RootPart.CFrame = fallbackPos
                Character:SetPrimaryPartCFrame(fallbackPos)
                Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                for _, x in ipairs(Character:GetChildren()) do
                    if x:IsA("BasePart") then
                        x.AssemblyLinearVelocity = Vector3.zero
                        x.AssemblyAngularVelocity = Vector3.zero
                        x.Velocity = Vector3.zero
                        x.RotVelocity = Vector3.zero
                    end
                end
            end
            task.wait()
        until not RootPart or (RootPart.Position - oldPos.p).Magnitude < 25 or tick() > returnTimeout

        Workspace.FallenPartsDestroyHeight = oldFPDH
        if RootPart then lastSafeCFrame = RootPart.CFrame end
        Humanoid:ChangeState(Enum.HumanoidStateType.Running)
    end)
    task.wait(0.04)
    flingInProgress = false
end

toggleAntiFling = function(state)
    antiFlingActive = state
    if antiFlingConn then antiFlingConn:Disconnect() antiFlingConn = nil end
    if antiFlingActive then
        antiFlingConn = RunService.Stepped:Connect(function()
            if not antiFlingActive then return end
            local myChar = LocalPlayer.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local pChar = player.Character
                    local pHRP = pChar:FindFirstChild("HumanoidRootPart")
                    for _, pPart in ipairs(pChar:GetDescendants()) do
                        if pPart:IsA("BasePart") and pPart.CanCollide then pPart.CanCollide = false end
                    end
                    if pHRP and myHRP then
                        local linVel = pHRP.AssemblyLinearVelocity.Magnitude
                        local angVel = pHRP.AssemblyAngularVelocity.Magnitude
                        local dist = (myHRP.Position - pHRP.Position).Magnitude
                        local isRealFling = (dist <= 7) and (angVel > 120 or linVel > 100)
                        if isRealFling then
                            for _, pPart in ipairs(pChar:GetDescendants()) do
                                if pPart:IsA("BasePart") then
                                    pPart.AssemblyLinearVelocity = Vector3.zero
                                    pPart.AssemblyAngularVelocity = Vector3.zero
                                    pPart.Velocity = Vector3.zero
                                    pPart.RotVelocity = Vector3.zero
                                end
                            end
                            local lastNotif = flingDebounceList[player] or 0
                            if (tick() - lastNotif) > 4 then
                                flingDebounceList[player] = tick()
                                Notify("Anti-Fling", "Fling bloqueado de: " .. player.DisplayName, 3.5)
                            end
                        end
                    end
                end
            end

            if myHRP and not isFlying and not flingInProgress and not autoFarmEnabled then
                if myHRP.AssemblyLinearVelocity.Magnitude > 120 then
                    myHRP.AssemblyLinearVelocity = Vector3.zero
                    myHRP.Velocity = Vector3.zero
                end
                if myHRP.AssemblyAngularVelocity.Magnitude > 120 then
                    myHRP.AssemblyAngularVelocity = Vector3.zero
                    myHRP.RotVelocity = Vector3.zero
                end
            end
        end)
    end
end

startFly = function()
    if isFlying then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    isFlying = true
    FlyInternal.bodyGyro = Instance.new("BodyGyro")
    FlyInternal.bodyGyro.P = 100000
    FlyInternal.bodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    FlyInternal.bodyGyro.CFrame = hrp.CFrame
    FlyInternal.bodyGyro.Parent = hrp

    FlyInternal.bodyVel = Instance.new("BodyVelocity")
    FlyInternal.bodyVel.P = 10000
    FlyInternal.bodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    FlyInternal.bodyVel.Velocity = Vector3.zero
    FlyInternal.bodyVel.Parent = hrp
    hum.PlatformStand = true

    FlyInternal.connection = RunService.Heartbeat:Connect(function()
        local character = LocalPlayer.Character
        if not isFlying or not character then
            if FlyInternal.connection then FlyInternal.connection:Disconnect() end
            return
        end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = character:FindFirstChild("HumanoidRootPart")
        local cam = Workspace.CurrentCamera
        if not humanoid or humanoid.Health <= 0 or not root or not cam then return end

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then
            currentFlySpeed = math.min(flyMaxSpeed, currentFlySpeed + flyAcceleration)
            FlyInternal.direction = moveDir.Unit
        else
            currentFlySpeed = math.max(0, currentFlySpeed - flyAcceleration)
        end

        local flatMove = Vector3.new(FlyInternal.direction.X, 0, FlyInternal.direction.Z)
        local velVector = Vector3.zero
        if flatMove.Magnitude > 0 then velVector = flatMove.Unit * currentFlySpeed end

        local lookVec = cam.CFrame.LookVector.Unit
        local dot = FlyInternal.direction:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(FlyInternal.direction:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * currentFlySpeed

        FlyInternal.bodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        FlyInternal.bodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec)
    end)
end

stopFly = function()
    if not isFlying then return end
    if FlyInternal.connection then FlyInternal.connection:Disconnect() FlyInternal.connection = nil end
    if FlyInternal.bodyGyro then FlyInternal.bodyGyro:Destroy() FlyInternal.bodyGyro = nil end
    if FlyInternal.bodyVel then FlyInternal.bodyVel:Destroy() FlyInternal.bodyVel = nil end
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
    end
    isFlying = false
    currentFlySpeed = 0
end

local function ApplyDrag(handle, frame)
    local isDragging = false
    local dragStart = Vector2.zero
    local startPos = UDim2.new()

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDragging = true
            dragStart = Vector2.new(input.Position.X, input.Position.Y)
            startPos = frame.Position

            local endConn
            endConn = UserInputService.InputEnded:Connect(function(endInput)
                if endInput.UserInputType == Enum.UserInputType.MouseButton1 or endInput.UserInputType == Enum.UserInputType.Touch then
                    isDragging = false
                    endConn:Disconnect()
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local floatStartPos = UDim2.new()

FloatingIcon.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isFloatDrag = true
        floatStartMouse = Vector2.new(input.Position.X, input.Position.Y)
        floatStartPos = FloatingIcon.Position

        local releaseConn
        releaseConn = UserInputService.InputEnded:Connect(function(endInput)
            if endInput.UserInputType == Enum.UserInputType.MouseButton1 or endInput.UserInputType == Enum.UserInputType.Touch then
                isFloatDrag = false
                releaseConn:Disconnect()
                if (Vector2.new(endInput.Position.X, endInput.Position.Y) - floatStartMouse).Magnitude < 10 then
                    ToggleHub(true)
                end
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isFloatDrag and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = Vector2.new(input.Position.X, input.Position.Y) - floatStartMouse
        FloatingIcon.Position = UDim2.new(floatStartPos.X.Scale, floatStartPos.X.Offset + delta.X, floatStartPos.Y.Scale, floatStartPos.Y.Offset + delta.Y)
    end
end)

    TabGrad.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0.00, Color3.fromRGB(60, 10, 20)),
        ColorSequenceKeypoint.new(0.45, Color3.fromRGB(35, 38, 45)),
        ColorSequenceKeypoint.new(0.50, Color3.fromRGB(255, 45, 75)),
        ColorSequenceKeypoint.new(0.55, Color3.fromRGB(35, 38, 45)),
        ColorSequenceKeypoint.new(1.00, Color3.fromRGB(60, 10, 20))
    })
    TabGrad.Parent = TabStroke
    table.insert(activeTabStrokeGradients, TabGrad)

    local TabLabel = Instance.new("TextLabel")
    TabLabel.Size = UDim2.new(1, -16, 1, 0)
    TabLabel.Position = UDim2.new(0, 10, 0, 0)
    TabLabel.BackgroundTransparency = 1
    TabLabel.Text = name
    TabLabel.TextColor3 = THEME.DimText
    TabLabel.Font = Enum.Font.GothamBold
    TabLabel.TextSize = 10
    TabLabel.TextXAlignment = Enum.TextXAlignment.Left
    TabLabel.Parent = TabBtn

    TabButtons[name] = {Btn = TabBtn, Label = TabLabel}

    local function SelectTab()
        if currentTab == name then return end
        currentTab = name
        for tabN, data in pairs(TabButtons) do
            local active = (tabN == name)
            Pages[tabN].Visible = active
            TweenService:Create(data.Btn, TweenInfo.new(0.2), {

            }):Play()
        end
    end

    if isDefault then SelectTab() end
    return Page
end

    end
end)

    local pct = math.clamp((currentVal - min) / (max - min), 0, 1)
    Fill.Size = UDim2.new(pct, 0, 1, 0)
    Fill.BackgroundColor3 = THEME.Crimson
    Fill.BorderSizePixel = 0
    Fill.Parent = Track

    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(1, 0)
    FillCorner.Parent = Fill

    local dragging = false
    local function UpdateSlider(xPos)
        local rel = math.clamp((xPos - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
        currentVal = math.floor(min + ((max - min) * rel))
        ValLabel.Text = tostring(currentVal)
        Fill.Size = UDim2.new(rel, 0, 1, 0)
        if callback then callback(currentVal) end
    end

    Card.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            UpdateSlider(input.Position.X)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            UpdateSlider(input.Position.X)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
