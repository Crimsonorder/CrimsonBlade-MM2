                    return Vector3.zero

        end))

            if not checkcaller() and typeof(self) == "Instance" and self:IsA("Humanoid") then

            end
            return oldNewIndex(self, key, value)
        end))

Core.getExactPingEjeT = function()
    if #Core.KinematicBuffer.pings == 0 then return 0.035 end
    local sum = 0
    for _, v in ipairs(Core.KinematicBuffer.pings) do sum = sum + v end
    return sum / #Core.KinematicBuffer.pings
end

Core.checkPlayerTool = function(player, toolName)
    if not player then return false end
    local bp = player:FindFirstChild("Backpack")
    local char = player.Character

    if bp then
        if bp:FindFirstChild(toolName) then return true end
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") and (t.Name == toolName or string.find(string.lower(t.Name), string.lower(toolName))) then
                return true
            end
        end
    end

    if char then
        if char:FindFirstChild(toolName) then return true end
        for _, t in ipairs(char:GetChildren()) do
            if t:IsA("Tool") and (t.Name == toolName or string.find(string.lower(t.Name), string.lower(toolName))) then
                return true
            end
        end
    end

    return false
end

Core.isPlayerInLobby = function()
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return true end

    local checkPos = hrp.Position
    if Core.isInvisible and Core.InvisInternal.controlBall and Core.InvisInternal.controlBall.Parent then
        checkPos = Core.InvisInternal.controlBall.Position
    end

    local map = Core.getMap()
    if not map then return true end

    local coinCont = map:FindFirstChild("CoinContainer")
    local spawns = map:FindFirstChild("Spawns")
    if not coinCont and not spawns then return true end

    if spawns and #spawns:GetChildren() > 0 then
        local closestDist = math.huge
        for _, sp in ipairs(spawns:GetChildren()) do
            if sp:IsA("BasePart") then
                local dist = (checkPos - sp.Position).Magnitude
                if dist < closestDist then closestDist = dist end
            end
        end
        if closestDist > 550 then return true end
    end

    if checkPos.Y > 120 and (checkPos - Vector3.new(-107, 152, 41)).Magnitude < 140 then
        return true
    end

    return false
end

Core.findGunHolder = function()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and Core.isPlayerAlive(player) and Core.checkPlayerTool(player, "Gun") then

Core.findNearestPlayer = function()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    if not Core.isPlayerAlive(LocalPlayer) then return nil end
    local myPos = LocalPlayer.Character.HumanoidRootPart.Position

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and Core.isPlayerAlive(player) then
            local otherRoot = player.Character.HumanoidRootPart
            local distance = (myPos - otherRoot.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                nearestPlayer = player
            end
        end
    end
    return nearestPlayer
end

Core.getPlayerByPartialName = function(input)
    if not input or input == "" then return nil end
    local query = string.lower(input:gsub("%s+", ""))
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local pName = string.lower(p.Name)
            local dName = string.lower(p.DisplayName)
            if string.sub(pName, 1, #query) == query or string.sub(dName, 1, #query) == query then
                return p
            end
        end
    end
    return nil
end

-- [[ 7. MOTOR BALÍSTICO EXTREMO V11 ]]
Core.checkUniversalLoS = function(origin, targetPos, targetChar)
    if not Core.InvisInternal.rayParams then
        Core.InvisInternal.rayParams = RaycastParams.new()
        Core.InvisInternal.rayParams.IgnoreWater = true
    end
    Core.InvisInternal.rayParams.FilterType = Enum.RaycastFilterType.Exclude
    Core.InvisInternal.rayParams.FilterDescendantsInstances = {LocalPlayer.Character, targetChar}

    local hit = Workspace:Raycast(origin, targetPos - origin, Core.InvisInternal.rayParams)
    if hit and hit.Instance and (hit.Instance.CanCollide == false or hit.Instance.Transparency > 0.7) then
        return true
    end
    return hit == nil
end

Core.getExtreme4AxisTarget = function(targetChar, origin)
    local hum = targetChar:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return nil end
    local targetBone = nil
    local bonesToScan = {"HumanoidRootPart", "UpperTorso", "Head", "LowerTorso"}
    
    if Core.consecutiveMisses > 0 or hum.FloorMaterial == Enum.Material.Air then
        bonesToScan = {
            "HumanoidRootPart", "UpperTorso", "Head", "LowerTorso",
            "RightUpperArm", "LeftUpperArm", "RightLowerLeg", "LeftLowerLeg", "RightFoot", "LeftFoot"
        }
    end
    for _, boneName in ipairs(bonesToScan) do
        local part = targetChar:FindFirstChild(boneName)
        if part and (Core.consecutiveMisses > 0 or Core.checkUniversalLoS(origin, part.Position, targetChar)) then
            targetBone = part
            break
        end
    end
    if not targetBone then
        targetBone = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("UpperTorso")
        if not targetBone then return nil end
    end
    local basePos = targetBone.Position
    local hrp = targetChar:FindFirstChild("HumanoidRootPart") or targetBone
    local dist = (basePos - origin).Magnitude
    local player = Players:GetPlayerFromCharacter(targetChar)
    local posBuf = player and Core.KinematicBuffer.positions[player]
    local vX, vY, vZ = 0, 0, 0
    local aX, aY, aZ = 0, 0, 0
    if posBuf and #posBuf >= 8 then
        local pNew = posBuf[#posBuf]
        local pMid = posBuf[#posBuf - 4]
        local pOld = posBuf[#posBuf - 7]
        local dT1 = pNew.t - pMid.t
        local dT2 = pMid.t - pOld.t
        if dT1 > 0 and dT2 > 0 then
            local vX1 = (pNew.pos.X - pMid.pos.X) / dT1
            local vY1 = (pNew.pos.Y - pMid.pos.Y) / dT1
            local vZ1 = (pNew.pos.Z - pMid.pos.Z) / dT1
            local vX2 = (pMid.pos.X - pOld.pos.X) / dT2
            local vY2 = (pMid.pos.Y - pOld.pos.Y) / dT2
            local vZ2 = (pMid.pos.Z - pOld.pos.Z) / dT2
            vX, vY, vZ = vX1, vY1, vZ1
            local totalDT = (pNew.t - pOld.t) * 0.5
            aX = (vX1 - vX2) / totalDT
            aY = (vY1 - vY2) / totalDT
            aZ = (vZ1 - vZ2) / totalDT
        end
    else
        local rawVel = hrp.AssemblyLinearVelocity
        vX, vY, vZ = rawVel.X, rawVel.Y, rawVel.Z
    end

    local pingEjeT = Core.getExactPingEjeT()
    local adaptativeRatio = (Core.consecutiveMisses > 0) and 1.30 or 1.10
    local flightTime = (dist / Core.BULLET_SPEED) + (pingEjeT * adaptativeRatio)

    if math.abs(vY) > 4 or hum.MoveDirection.Magnitude > 0.1 then
        aX = aX * 1.25
        aZ = aZ * 1.25
        aY = aY * 1.40
    end

    local leadX = (vX * flightTime) + (0.5 * aX * (flightTime ^ 2))
    local leadZ = (vZ * flightTime) + (0.5 * aZ * (flightTime ^ 2))
    local leadY = 0
    local inAir = (hum.FloorMaterial == Enum.Material.Air) or (math.abs(vY) > 1.0)
    if inAir then
        local g = Workspace.Gravity > 0 and Workspace.Gravity or 196.2
        leadY = (vY * flightTime) + (0.5 * (aY - g) * (flightTime ^ 2))
    else
        leadY = (vY * flightTime) + (0.5 * aY * (flightTime ^ 2))
    end
    local finalLeadVector = Vector3.new(leadX, leadY, leadZ)
    local maxMove = (hum.WalkSpeed > 0 and hum.WalkSpeed or 16) * flightTime * 1.5
    if finalLeadVector.Magnitude > maxMove then
        finalLeadVector = finalLeadVector.Unit * maxMove
    end
    return basePos + finalLeadVector
end

Core.runExtremeShootAction = function(silent)
    if Core.shootActionDebounce then return false end
    Core.shootActionDebounce = true

    local char = LocalPlayer.Character
    if not char then Core.shootActionDebounce = false return false end

    local hum = char:FindFirstChildOfClass("Humanoid")
    local myHRP = char:FindFirstChild("HumanoidRootPart")
    if not hum or not myHRP or hum.Health <= 0 then
        Core.shootActionDebounce = false

    end

    local backpack = LocalPlayer:FindFirstChild("Backpack")

    if not gun then

        Core.shootActionDebounce = false
        return false
    end

    if not gun.ManualActivationOnly then
        pcall(function() gun.ManualActivationOnly = true end)
    end

    if gun.Parent == backpack then
        gun.Parent = char
        task.wait(0.02)
    end

    local murderer = Core.cachedMurderer
    if not murderer or not murderer.Character or not murderer.Character:FindFirstChild("Humanoid") or murderer.Character.Humanoid.Health <= 0 then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
                if (player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Knife")) or player.Character:FindFirstChild("Knife") then
                    murderer = player
                    Core.cachedMurderer = player
                    break
                end
            end
        end
    end

    if not murderer or not murderer.Character then
        if not silent and Core.Notify then Core.Notify("Shoot", "Murderer no detectado en ronda.", 2) end
        Core.shootActionDebounce = false
        return false
    end

    local mHRP = murderer.Character:FindFirstChild("HumanoidRootPart") or murderer.Character:FindFirstChild("UpperTorso")
    if not mHRP then
        Core.shootActionDebounce = false
        return false
    end

    local realOrigin = myHRP.Position + (myHRP.CFrame.LookVector * 1.2)
    local targetPos = Core.getExtreme4AxisTarget(murderer.Character, realOrigin)

    if not targetPos then
        targetPos = mHRP.Position
        Core.consecutiveMisses = Core.consecutiveMisses + 1
    else
        Core.consecutiveMisses = 0
    end

    local finalOrigin = realOrigin
    if not Core.checkUniversalLoS(realOrigin, targetPos, murderer.Character) then
        if Core.quantumWallPierce then
            finalOrigin = targetPos - (murderer.Character.HumanoidRootPart.CFrame.LookVector * 1.5)
        else
            if not silent and Core.Notify then
                Core.Notify("Apex Shoot", "Murderer protegido por pared.", 2.5)
            end
            Core.shootActionDebounce = false
            return false
        end
    end

    pcall(function()
        local lookAtTarget = CFrame.lookAt(myHRP.Position, Vector3.new(targetPos.X, myHRP.Position.Y, targetPos.Z))
        myHRP.CFrame = lookAtTarget
        Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, targetPos)
    end)

    local aimLookCFrame = CFrame.lookAt(finalOrigin, targetPos)
    local targetCFrame = CFrame.new(targetPos)

    local shootRemote = gun:FindFirstChild("Shoot")
    if shootRemote then
        shootRemote:FireServer(aimLookCFrame, targetCFrame)
    end

    task.spawn(function()
        pcall(function()
            if gun:FindFirstChild("KnifeLocal") and gun.KnifeLocal:FindFirstChild("CreateBeam") and gun.KnifeLocal.CreateBeam:FindFirstChild("RemoteFunction") then
                gun.KnifeLocal.CreateBeam.RemoteFunction:InvokeServer(1, targetPos, "AH2")
            end
        end)
    end)

    if not silent and Core.Notify then
        if Core.quantumWallPierce and not Core.checkUniversalLoS(realOrigin, targetPos, murderer.Character) then
            Core.Notify("Apex Shoot", "Bypass Cuántico: ¡Tiro atravesó el muro!", 3)
        else
            Core.Notify("Apex Shoot", "Disparo interceptado con éxito.", 2.5)
        end
    end

    task.delay(0.04, function() Core.shootActionDebounce = false end)
    return true
end

local function getMapSpawnInvis()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("SpawnLocation") and not string.find(string.lower(obj.Name), "lobby") and obj.Position.Y < 100 then
            return obj.CFrame
        end
    end
    local map = Core.getMap()
    if map then
        local spawns = map:FindFirstChild("Spawns") or map:FindFirstChild("SpawnLocation") or map:FindFirstChild("Spawn")
        if spawns then
            if spawns:IsA("BasePart") then return spawns.CFrame end
            local p = spawns:FindFirstChildWhichIsA("BasePart") or spawns:FindFirstChildOfClass("SpawnLocation")
            if p then return p.CFrame end
        end
        local anyPart = map:FindFirstChildWhichIsA("BasePart")
        if anyPart and anyPart.Position.Y < 100 then
            return anyPart.CFrame + Vector3.new(0, 4, 0)
        end
    end
    return nil
end

local function checkStairsRayInvis(ballPos, moveDir)
    if moveDir.Magnitude == 0 then return false end
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, Core.InvisInternal.controlBall, Core.InvisInternal.skyPlatform}
    rayParams.IgnoreWater = true
    local origin = ballPos + Vector3.new(0, -1.2, 0)
    local direction = moveDir.Unit * 2.2
    local hit = Workspace:Raycast(origin, direction, rayParams)
    if hit and hit.Instance and hit.Instance.CanCollide then
        local upperOrigin = ballPos + Vector3.new(0, 1.2, 0)
        local upperHit = Workspace:Raycast(upperOrigin, direction, rayParams)
        if not upperHit then return true end
    end
    return false
end

local function checkClimbRayInvis(ballPos, moveDir)
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {LocalPlayer.Character, Core.InvisInternal.controlBall, Core.InvisInternal.skyPlatform}
    rayParams.IgnoreWater = true
    local scanDir = moveDir.Magnitude > 0 and moveDir.Unit or Vector3.new(Camera.CFrame.LookVector.X, 0, Camera.CFrame.LookVector.Z).Unit
    local hit = Workspace:Raycast(ballPos, scanDir * 2.2, rayParams)
    if hit and hit.Instance and hit.Instance.CanCollide then
        if not string.find(string.lower(hit.Instance.Name), "floor") and not string.find(string.lower(hit.Instance.Name), "ground") then
            return true
        end
    end
    return false
end

Core.toggleInvisibility = function()
    local realChar = LocalPlayer.Character
    local realHRP = realChar and realChar:FindFirstChild("HumanoidRootPart")

    if not realHRP or not realHum or realHum.Health <= 0 then return end

        if Core.Notify then
            Core.Notify("Invisibilidad", "No puedes usar invisibilidad en el Lobby. Espera la ronda.", 3)
        end
        return
    end

    Core.isInvisible = not Core.isInvisible

    if Core.isInvisible then
        setupInvisibilityEngine(nil)

        if Core.ghostButtonInstance then
            Core.ghostButtonInstance.Text = "Invisible: ON"
            Core.ghostButtonInstance.TextColor3 = Color3.fromRGB(80, 255, 120)

        end
        if Core.Notify then Core.Notify("Ghost Engine", "Invisibilidad Sky Ball ACTIVADA.", 3) end
    else
        local finalLocation = nil
        if Core.InvisInternal.controlBall then
            finalLocation = Core.InvisInternal.controlBall.CFrame
        end

        clearGhostInstancesEngine()

        if realHRP and finalLocation then
            realHRP.Anchored = false
            realHRP.CFrame = finalLocation + Vector3.new(0, 2, 0)
            realHRP.AssemblyLinearVelocity = Vector3.zero
            realHRP.AssemblyAngularVelocity = Vector3.zero
        elseif realHRP and Core.isPlayerInLobby() then
            realHRP.Anchored = false
            realHRP.CFrame = CFrame.new(-107, 152, 41)
            realHRP.AssemblyLinearVelocity = Vector3.zero
            realHRP.AssemblyAngularVelocity = Vector3.zero
        end

        if realChar then
            for _, part in ipairs(realChar:GetDescendants()) do
                if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                    part.Transparency = 0
                elseif part:IsA("Decal") then
                    part.Transparency = 0
                end
            end
        end

        Camera.CameraSubject = realHum
        Camera.CameraType = Enum.CameraType.Custom
        if realHum then
            realHum:ChangeState(Enum.HumanoidStateType.GettingUp)
        end

        if Core.ghostButtonInstance then
            Core.ghostButtonInstance.Text = "Invisibility"
            Core.ghostButtonInstance.TextColor3 = Core.THEME.SilverText
            Core.applyTextLaser(Core.ghostButtonInstance, false)
        end
        if Core.Notify then Core.Notify("Ghost Engine", "Invisibilidad DESACTIVADA.", 2.5) end
    end
end

Core.startFly = function()
    if Core.isFlying then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    Core.isFlying = true
    Core.FlyInternal.bodyGyro = Instance.new("BodyGyro")
    Core.FlyInternal.bodyGyro.P = 100000
    Core.FlyInternal.bodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    Core.FlyInternal.bodyGyro.CFrame = hrp.CFrame
    Core.FlyInternal.bodyGyro.Parent = hrp

    Core.FlyInternal.bodyVel = Instance.new("BodyVelocity")
    Core.FlyInternal.bodyVel.P = 10000
    Core.FlyInternal.bodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    Core.FlyInternal.bodyVel.Velocity = Vector3.zero
    Core.FlyInternal.bodyVel.Parent = hrp
    hum.PlatformStand = true

    Core.flyNoclipConn = RunService.Stepped:Connect(function()
        if Core.isFlying and LocalPlayer.Character then
            for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
    end)

    Core.FlyInternal.connection = RunService.Heartbeat:Connect(function()
        local character = LocalPlayer.Character
        if not Core.isFlying or not character then
            if Core.FlyInternal.connection then Core.FlyInternal.connection:Disconnect() end
            return
        end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = character:FindFirstChild("HumanoidRootPart")
        local cam = Workspace.CurrentCamera
        if not humanoid or humanoid.Health <= 0 or not root or not cam then return end

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then
            Core.currentFlySpeed = math.min(Core.flyMaxSpeed, Core.currentFlySpeed + Core.flyAcceleration)
            Core.FlyInternal.direction = moveDir.Unit
        else
            Core.currentFlySpeed = math.max(0, Core.currentFlySpeed - Core.flyAcceleration)
        end

        local flatMove = Vector3.new(Core.FlyInternal.direction.X, 0, Core.FlyInternal.direction.Z)
        local velVector = Vector3.zero
        if flatMove.Magnitude > 0 then velVector = flatMove.Unit * Core.currentFlySpeed end

        local lookVec = cam.CFrame.LookVector.Unit
        local dot = Core.FlyInternal.direction:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(Core.FlyInternal.direction:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * Core.currentFlySpeed

        Core.FlyInternal.bodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        Core.FlyInternal.bodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec)
    end)
end

Core.stopFly = function()
    if not Core.isFlying then return end
    if Core.FlyInternal.connection then Core.FlyInternal.connection:Disconnect() Core.FlyInternal.connection = nil end
    if Core.FlyInternal.bodyGyro then Core.FlyInternal.bodyGyro:Destroy() Core.FlyInternal.bodyGyro = nil end
    if Core.FlyInternal.bodyVel then Core.FlyInternal.bodyVel:Destroy() Core.FlyInternal.bodyVel = nil end
    if Core.flyNoclipConn then Core.flyNoclipConn:Disconnect() Core.flyNoclipConn = nil end

    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                part.CanCollide = true
            end
        end
    end
    Core.isFlying = false
    Core.currentFlySpeed = 0
end

local function isRealVoid(pos)
    local p = RaycastParams.new()
    p.FilterType = Enum.RaycastFilterType.Exclude
    p.FilterDescendantsInstances = {LocalPlayer.Character, Core.airWalkPlatform}
    p.IgnoreWater = false
    local hit = Workspace:Raycast(pos, Vector3.new(0, -350, 0), p)
    return hit == nil
end

Core.toggleVoidWalk = function(enable)

    if Core.airWalkConnection then Core.airWalkConnection:Disconnect() Core.airWalkConnection = nil end

            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                    part.CanCollide = true
                end
            end
        end
        return
    end

    Core.noclipConnection = RunService.Stepped:Connect(function()
        if Core.airWalkEnabled and LocalPlayer.Character then
            for _, child in ipairs(LocalPlayer.Character:GetDescendants()) do
                if child:IsA("BasePart") then child.CanCollide = false end
            end
        end
    end)

    local lastAirWalkTick = 0
    Core.airWalkConnection = RunService.RenderStepped:Connect(function()
        if not Core.scriptActive or not Core.airWalkEnabled or Core.isInvisible or Core.isFlying then
            if Core.airWalkPlatform then Core.airWalkPlatform:Destroy() Core.airWalkPlatform = nil end
            return
        end

        local now = os.clock()
        if now - lastAirWalkTick < 0.04 then return end
        lastAirWalkTick = now

        local curChar = LocalPlayer.Character
        local hrp = curChar and curChar:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        if isRealVoid(hrp.Position) then
            if not Core.airWalkPlatform or not Core.airWalkPlatform.Parent then

                Core.airWalkPlatform.Name = "VoidWalkSolidPlatform"
                Core.airWalkPlatform.Size = Vector3.new(8, 1, 8)
                Core.airWalkPlatform.Transparency = 1
                Core.airWalkPlatform.Anchored = true
                Core.airWalkPlatform.CanCollide = true
                Core.airWalkPlatform.Parent = Workspace
            end
            Core.airWalkPlatform.Position = Vector3.new(hrp.Position.X, hrp.Position.Y - 3.1, hrp.Position.Z)
        else
            if Core.airWalkPlatform then
                Core.airWalkPlatform:Destroy()
                Core.airWalkPlatform = nil
            end
        end
    end)
end

Core.findMurderer = function()
    for _, player in ipairs(Players:GetPlayers()) do
        if Core.isPlayerAlive(player) and Core.checkPlayerTool(player, "Knife") then
            return player
        end
    end
    if Core.playerDataESP then
        for player, data in pairs(Core.playerDataESP) do
            if (data.Role == "Murderer" or data.role == "Murderer") and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if Core.isPlayerAlive(p) then return p end
            end
        end
    end
    if Core.gamePlayerData then
        for player, data in pairs(Core.gamePlayerData) do
            if (data.Role == "Murderer" or data.role == "Murderer") and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if Core.isPlayerAlive(p) then return p end
            end
        end
    end
    return nil
end

Core.applyHighlightToCharacter = function(player, character)
    if not Core.espActive or not character or player == LocalPlayer then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")

    if not humanoid or humanoid.Health <= 0 then
        if Core.ESPInternal.billboards[character] then Core.ESPInternal.billboards[character]:Destroy() Core.ESPInternal.billboards[character] = nil end
        if Core.ESPInternal.highlights[character] then Core.ESPInternal.highlights[character]:Destroy() Core.ESPInternal.highlights[character] = nil end
        return
    end

    local hasKnife = Core.checkPlayerTool(player, "Knife")
    local hasGun = Core.checkPlayerTool(player, "Gun")
    local isMurderer = hasKnife or (Core.cachedMurderer == player) or (Core.findMurderer() == player)
    local isSheriff = false
    local isHero = false

    if not isMurderer then
        if hasGun then
            local originalRole = (Core.playerDataESP[player.Name] and (Core.playerDataESP[player.Name].Role or Core.playerDataESP[player.Name].role))
                or (Core.gamePlayerData[player.Name] and (Core.gamePlayerData[player.Name].Role or Core.gamePlayerData[player.Name].role))
            if originalRole and originalRole ~= "Sheriff" then isHero = true else isSheriff = true end
        else
            local sh = Core.findSheriff()
            if sh == player then isSheriff = true end
        end
    end

    if Core.ignoreInnocents and not isMurderer and not isSheriff and not isHero then
        if Core.ESPInternal.highlights[character] then Core.ESPInternal.highlights[character]:Destroy() Core.ESPInternal.highlights[character] = nil end
        if Core.ESPInternal.billboards[character] then Core.ESPInternal.billboards[character]:Destroy() Core.ESPInternal.billboards[character] = nil end
        return
    end

    local hl = Core.ESPInternal.highlights[character]
    if not hl or not hl.Parent then
        if hl then hl:Destroy() end
        hl = Instance.new("Highlight")
        hl.Name = "Player_Highlight"
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 1
        hl.OutlineTransparency = 0
        hl.Adornee = character
        hl.Parent = Core.PARENT
        Core.ESPInternal.highlights[character] = hl
    end

    local targetColor = Color3.fromRGB(0, 255, 8)
    local roleText = nil
    local textColor = Color3.fromRGB(255, 255, 255)

    if isMurderer then
        targetColor = Color3.fromRGB(255, 40, 40)
        roleText = "MURDERER"
        textColor = Color3.fromRGB(255, 85, 85)
    elseif isSheriff then
        targetColor = Color3.fromRGB(40, 140, 255)
        roleText = "SHERIFF"
        textColor = Color3.fromRGB(80, 185, 255)
    elseif isHero then
        targetColor = Color3.fromRGB(0, 210, 255)
        roleText = "HERO"
        textColor = Color3.fromRGB(95, 220, 255)
    end

    if hl.OutlineColor ~= targetColor then hl.OutlineColor = targetColor end
    hl.FillTransparency = 1
    hl.OutlineTransparency = 0
    hl.Enabled = true

    if roleText and head then
        local bg = Core.ESPInternal.billboards[character]
        if not bg or not bg.Parent then
            if bg then bg:Destroy() end
            bg = Instance.new("BillboardGui")
            bg.Name = "Role_Billboard"
            bg.AlwaysOnTop = true
            bg.Size = UDim2.new(0, 140, 0, 30)
            bg.StudsOffset = Vector3.new(0, 2.5, 0)
            bg.Adornee = head
            bg.Parent = Core.PARENT

            local lbl = Instance.new("TextLabel")
            lbl.Name = "RoleLabel"
            lbl.Size = UDim2.new(1, 0, 1, 0)
            lbl.BackgroundTransparency = 1
            lbl.Font = Enum.Font.GothamBold
            lbl.TextSize = 13
            lbl.Parent = bg

            local stroke = Instance.new("UIStroke")
            stroke.Thickness = 1.5
            stroke.Color = Color3.fromRGB(0, 0, 0)
            stroke.Parent = lbl
            Core.ESPInternal.billboards[character] = bg
        end
        local lbl = bg:FindFirstChild("RoleLabel")
        if lbl then
            lbl.Text = roleText
            lbl.TextColor3 = textColor
        end
    else
        if Core.ESPInternal.billboards[character] then
            Core.ESPInternal.billboards[character]:Destroy()
            Core.ESPInternal.billboards[character] = nil
        end
    end
end

Core.reloadESP = function()
    if not Core.espActive then Core.clearESP() Core.updateGunDropESP() return end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then Core.applyHighlightToCharacter(player, player.Character) end
    end
    Core.updateGunDropESP()
end

Core.startESP = function()
    Core.espActive = true
    local globalRemotes = ReplicatedStorage:WaitForChild("Remotes", 5)
    if globalRemotes and globalRemotes:FindFirstChild("Gameplay") and globalRemotes.Gameplay:FindFirstChild("PlayerDataChanged") then
        Core.databaseConnectionESP = globalRemotes.Gameplay.PlayerDataChanged.OnClientEvent:Connect(function(data)
            Core.playerDataESP = data
            Core.reloadESP()
        end)
    end

    local function setupPlayerESP(player)
        if player == LocalPlayer then return end
        if player.Character then task.spawn(Core.applyHighlightToCharacter, player, player.Character) end

        local function attachToolListener(container)
            if not container then return end
            container.ChildAdded:Connect(function(c)
                if c:IsA("Tool") and (c.Name == "Knife" or c.Name == "Gun") then
                    task.wait(0.02)
                    Core.reloadESP()
                end
            end)
        end

        local pConn = player.CharacterAdded:Connect(function(character)
            task.wait(0.04)
            attachToolListener(character)
            if Core.espActive then Core.applyHighlightToCharacter(player, character) end
        end)

        task.spawn(function()
            local bp = player:WaitForChild("Backpack", 3)
            attachToolListener(bp)
            if player.Character then attachToolListener(player.Character) end
        end)

        if not Core.playerConnections[player] then Core.playerConnections[player] = {} end
        table.insert(Core.playerConnections[player], pConn)
    end

    Core.playerAddedConnectionESP = Players.PlayerAdded:Connect(setupPlayerESP)
    for _, player in ipairs(Players:GetPlayers()) do setupPlayerESP(player) end

    Core.espLoop = task.spawn(function()
        while Core.espActive and Core.scriptActive do
            Core.reloadESP()
            task.wait(0.3)
        end
    end)
    Core.reloadESP()
end

Core.stopESP = function()
    Core.espActive = false
    if Core.espLoop then task.cancel(Core.espLoop) Core.espLoop = nil end
    Core.clearESP()
    if Core.databaseConnectionESP then Core.databaseConnectionESP:Disconnect() Core.databaseConnectionESP = nil end
    if Core.playerAddedConnectionESP then Core.playerAddedConnectionESP:Disconnect() Core.playerAddedConnectionESP = nil end
    for _, conns in pairs(Core.playerConnections) do
        for _, conn in ipairs(conns) do
            pcall(function() conn:Disconnect() end)
        end
    end
    table.clear(Core.playerConnections)
end

Core.executeKillTarget = function(targetInput)
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local knife = char and (char:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife")))

    if not knife then
        if Core.Notify then Core.Notify("Error", "No tienes el cuchillo equipado o no eres Murderer.", 2.5) end
        return
    end

    local hum = char:FindFirstChildOfClass("Humanoid")
    local myHRP = char:FindFirstChild("HumanoidRootPart")
    if not myHRP or not hum or hum.Health <= 0 then return end

    local targetPlayer = Core.getPlayerByPartialName(targetInput)
    if not targetPlayer or not targetPlayer.Character then
        if Core.Notify then Core.Notify("Kill Target", "Jugador objetivo no encontrado.", 2.5) end
        return
    end

    -- Validación estricta: ¿Está vivo y fuera del lobby?
    if not Core.isPlayerAlive(targetPlayer) or Core.isPlayerInLobby() then
        if Core.Notify then Core.Notify("Kill Target", "El objetivo está muerto o en el Lobby.", 2.5) end
        return
    end

    local tHRP = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not tHRP then return end

    if knife.Parent == backpack and hum then hum:EquipTool(knife) end

    tHRP.CFrame = myHRP.CFrame + (myHRP.CFrame.LookVector * 1.5)
    task.wait(0.05)
    if char:FindFirstChild("Knife") and char.Knife:FindFirstChild("Stab") then
        char.Knife.Stab:FireServer("Slash")
        if Core.Notify then Core.Notify("Eliminado", "Eliminaste a: " .. targetPlayer.DisplayName, 3) end
    end
end

Core.runKnifeThrowNATIVE = function(silent)
    if Core.knifeActionDebounce then return end
    Core.knifeActionDebounce = true

    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local knife = char:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife"))
        if not knife or not hum or hum.Health <= 0 then
            if not silent and Core.Notify then Core.Notify("Error", "No tienes el cuchillo equipado.", 2) end
            return
        end

        if knife.Parent == backpack then
            hum:EquipTool(knife)
            task.wait(0.02)
        end

        local target = Core.findNearestPlayer()
        if not target or not target.Character or not Core.isPlayerAlive(target) or Core.isPlayerInLobby() then
            if not silent and Core.Notify then Core.Notify("Error", "No hay objetivos vivos válidos cercanos.", 2) end
            return
        end

        local tHRP = target.Character:FindFirstChild("HumanoidRootPart")
        local myHRP = char:FindFirstChild("HumanoidRootPart")
        local rightHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or myHRP
        if not tHRP or not myHRP or not rightHand then return end

        local targetDirectPos = Core.getExtreme4AxisTarget(target.Character, rightHand.Position) or (tHRP.Position + Vector3.new(0, 1, 0))
        local dist = (myHRP.Position - tHRP.Position).Magnitude

        if dist <= 8 and knife:FindFirstChild("Stab") then knife.Stab:FireServer("Slash") end

        local knifeEvents = knife:FindFirstChild("Events")
        local knifeThrownRemote = knifeEvents and knifeEvents:FindFirstChild("KnifeThrown")
        local throwRemote = knife:FindFirstChild("Throw")
        local argsThrow = { CFrame.new(rightHand.Position), CFrame.new(targetDirectPos) }

        if knifeThrownRemote then
            knifeThrownRemote:FireServer(unpack(argsThrow))
        elseif throwRemote then
            throwRemote:FireServer(unpack(argsThrow))
        end
    end)
    task.wait(0.14)
    Core.knifeActionDebounce = false
end

-- [[ FLING ULTRA POTENTE CON INMUNIDAD TEMPORAL Y RETORNO SEGURO ]]
Core.executeFlingAction = function(TargetPlayer, customReturnCFrame)
    if not TargetPlayer or TargetPlayer == LocalPlayer or Core.flingInProgress then return end
    if not Core.isPlayerAlive(LocalPlayer) or not Core.isPlayerAlive(TargetPlayer) or Core.isPlayerInLobby() then return end

    Core.flingInProgress = true
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart

    local TCharacter = TargetPlayer.Character
    local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
    local TRootPart = THumanoid and THumanoid.RootPart
    local THead = TCharacter and TCharacter:FindFirstChild("Head")

    if not Character or not Humanoid or not RootPart or not TCharacter or not THumanoid then
        Core.flingInProgress = false
        return
    end

    local oldPos = customReturnCFrame or RootPart.CFrame
    local oldFPDH = Workspace.FallenPartsDestroyHeight
    local Cam = Workspace.CurrentCamera

    pcall(function()
        if RootPart.Velocity.Magnitude < 50 then getgenv().OldPos = oldPos end
        if THead then Cam.CameraSubject = THead elseif THumanoid then Cam.CameraSubject = THumanoid end
        if not TCharacter:FindFirstChildWhichIsA("BasePart") then return end

        local function FPos(BasePart, Pos, Ang)
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
            RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
            RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        end

        local function SFBasePart(BasePart)
            local TimeToWait = 2.5
            local Time = tick()
            local Angle = 0

            repeat
                if RootPart and THumanoid then
                    Angle = Angle + 100
                    FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                    FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                    task.wait()
                else
                    break
                end
            until BasePart.Velocity.Magnitude > 500 or BasePart.Position.Y < -40 or BasePart.Parent ~= TargetPlayer.Character or TargetPlayer.Parent ~= Players or TargetPlayer.Character ~= TCharacter or THumanoid.Sit or Humanoid.Health <= 0 or tick() > Time + TimeToWait
        end

        Workspace.FallenPartsDestroyHeight = 0/0
        local BV = Instance.new("BodyVelocity")
        BV.Name = "EpixVel"
        BV.Parent = RootPart
        BV.Velocity = Vector3.new(9e8, 9e8, 9e8)
        BV.MaxForce = Vector3.new(1/0, 1/0, 1/0)
        
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)

        if TRootPart then SFBasePart(TRootPart) elseif THead then SFBasePart(THead) end

        BV:Destroy()
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
        Cam.CameraSubject = Humanoid

        local returnTimeout = tick() + 0.40
        repeat
            if RootPart and Character then
                local fallbackPos = oldPos * CFrame.new(0, .5, 0)
                RootPart.CFrame = fallbackPos
                Character:SetPrimaryPartCFrame(fallbackPos)
                Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                for _, x in ipairs(Character:GetChildren()) do
                    if x:IsA("BasePart") then
                        x.AssemblyLinearVelocity = Vector3.zero
                        x.AssemblyAngularVelocity = Vector3.zero
                        x.Velocity = Vector3.zero
                        x.RotVelocity = Vector3.zero
                    end
                end
            end
            task.wait()
        until not RootPart or (RootPart.Position - oldPos.p).Magnitude < 25 or tick() > returnTimeout

        Workspace.FallenPartsDestroyHeight = oldFPDH
        if RootPart then Core.lastSafeCFrame = RootPart.CFrame end
        Humanoid:ChangeState(Enum.HumanoidStateType.Running)
    end)
    task.wait(0.04)
    Core.flingInProgress = false
end

Core.toggleAntiFling = function(state)
    Core.antiFlingActive = state
    if Core.antiFlingConn then Core.antiFlingConn:Disconnect() Core.antiFlingConn = nil end
    if Core.antiFlingActive then
        Core.antiFlingConn = RunService.Stepped:Connect(function()
            if not Core.antiFlingActive then return end
            local myChar = LocalPlayer.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local pChar = player.Character
                    local pHRP = pChar:FindFirstChild("HumanoidRootPart")
                    for _, pPart in ipairs(pChar:GetDescendants()) do
                        if pPart:IsA("BasePart") and pPart.CanCollide then pPart.CanCollide = false end
                    end
                    if pHRP and myHRP then
                        local linVel = pHRP.AssemblyLinearVelocity.Magnitude
                        local angVel = pHRP.AssemblyAngularVelocity.Magnitude
                        local dist = (myHRP.Position - pHRP.Position).Magnitude
                        local isRealFling = (dist <= 7) and (angVel > 120 or linVel > 100)
                        if isRealFling then
                            for _, pPart in ipairs(pChar:GetDescendants()) do
                                if pPart:IsA("BasePart") then
                                    pPart.AssemblyLinearVelocity = Vector3.zero
                                    pPart.AssemblyAngularVelocity = Vector3.zero
                                    pPart.Velocity = Vector3.zero
                                    pPart.RotVelocity = Vector3.zero
                                end
                            end
                            local lastNotif = Core.flingDebounceList[player] or 0
                            if (tick() - lastNotif) > 4 then
                                Core.flingDebounceList[player] = tick()
                                if Core.Notify then Core.Notify("Anti-Fling", "Fling bloqueado de: " .. player.DisplayName, 3.5) end
                            end
                        end
                    end
                end
            end

            if myHRP and not Core.isFlying and not Core.flingInProgress and not Core.autoFarmEnabled then
                if myHRP.AssemblyLinearVelocity.Magnitude > 120 then
                    myHRP.AssemblyLinearVelocity = Vector3.zero
                    myHRP.Velocity = Vector3.zero
                end
                if myHRP.AssemblyAngularVelocity.Magnitude > 120 then
                    myHRP.AssemblyAngularVelocity = Vector3.zero
                    myHRP.RotVelocity = Vector3.zero
                end
            end
        end)
    end
end

Core.farmObtenerSpawnMapa = function()
    local map = Core.getMap()
    if map and map:FindFirstChild("Spawns") then
        local spawns = map.Spawns:GetChildren()
        if #spawns > 0 then return spawns[math.random(1, #spawns)].CFrame + Vector3.new(0, 3.5, 0) end
    end
    for _, part in ipairs(Workspace:GetDescendants()) do
        if part:IsA("SpawnLocation") and not string.find(string.lower(part.Name), "lobby") then
            return part.CFrame + Vector3.new(0, 3.5, 0)
        end
    end

end

Core.farmVolverASuperficie = function(hrp)
    if not hrp then return end
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    local sp = Core.farmObtenerSpawnMapa()
    if sp then
        hrp.CFrame = sp
    else
        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        rayParams.FilterDescendantsInstances = {LocalPlayer.Character}
        rayParams.IgnoreWater = true
        local upRay = Workspace:Raycast(hrp.Position, Vector3.new(0, 60, 0), rayParams)
        if upRay and upRay.Position then
            hrp.CFrame = CFrame.new(upRay.Position + Vector3.new(0, 4.0, 0))
        else
            hrp.CFrame = hrp.CFrame + Vector3.new(0, Core.PROFUNDIDAD_SUELO + 4.5, 0)
        end
    end
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    Core.farmIsUnderground = false
end

Core.farmObtenerSiguienteMoneda = function(hrp)
    if not hrp then return nil end
    local mejorMoneda = nil
    local menorDistancia = math.huge
    local ahora = os.clock()

    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name == "Coin_Server" and obj.Parent ~= LocalPlayer.Character then
            local tieneVisual = obj:FindFirstChild("CoinVisual") or obj:FindFirstChild("Coin")
            if tieneVisual then
                if not Core.FarmInternal.ignoredCoins[obj] or ahora > Core.FarmInternal.ignoredCoins[obj] then
                    local dist = (hrp.Position - obj.Position).Magnitude
                    if dist < menorDistancia then
                        menorDistancia = dist
                        mejorMoneda = obj
                    end
                end
            end
        end
    end
    return mejorMoneda
end

Core.farmEjecutarLoop = function()
    while Core.autoFarmEnabled and Core.scriptActive do
        local character = LocalPlayer.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")

        if hrp and humanoid and humanoid.Health > 0 then
            if Core.isPlayerInLobby() then
                Core.farmStatusString = "[STANDBY - LOBBY IDLE]"
                Core.farmEnSpawn = false
                Core.farmIsUnderground = false
                Core.farmDesactivarNoclip()
                Core.farmDesactivarXRay()
                Core.farmRestaurarCamara()
                Core.farmRestaurarFisicas(humanoid)
                task.wait(0.8)
            else
                if Core.bagCoinsActual >= Core.bagCoinsMax and Core.bagCoinsMax > 0 then
                    Core.farmStatusString = "[REPOSITORY FULL - SECURING]"
                    if not Core.farmEnSpawn then
                        Core.farmEnSpawn = true
                        Core.farmDesactivarNoclip()
                        Core.farmDesactivarXRay()
                        Core.farmRestaurarCamara()
                        Core.farmRestaurarFisicas(humanoid)

                        local sp = Core.farmObtenerSpawnMapa() or (hrp.CFrame + Vector3.new(0, 5, 0))
                        hrp.CFrame = sp
                        Core.farmIsUnderground = false
                        task.wait(0.25)

                        local amIMurderer = Core.checkPlayerTool(LocalPlayer, "Knife")
                        if amIMurderer then
                            if Core.lethalMurdererPurgeEnabled and not Core.farmFlingHechoRonda then
                                if Core.Notify then Core.Notify("Auto-Farm", "Bolsa llena (40/40). Ejecutando Kill Everyone nativo...", 3) end
                                Core.executeKillEveryoneNative()
                                Core.farmFlingHechoRonda = true
                            else
                                if Core.Notify then Core.Notify("Auto-Farm", "Bolsa llena (40/40). En zona segura como Murderer.", 3) end
                            end
                        else
                            if Core.autoFlingOnFullEnabled and not Core.farmFlingHechoRonda then
                                local murd = Core.findMurderer()
                                if murd then
                                    if Core.Notify then Core.Notify("Auto-Farm", "Bolsa llena (40/40). Fling a Murderer iniciado...", 3) end
                                    Core.executeFlingAction(murd, sp)
                                    Core.farmFlingHechoRonda = true
                                end
                            else
                                if Core.Notify then Core.Notify("Auto-Farm", "Bolsa llena (40/40). Manteniendo posición segura.", 3) end
                            end
                        end
                    end
                    task.wait(1)
                else
                    local moneda = Core.farmObtenerSiguienteMoneda(hrp)
                    if moneda and moneda.Parent then
                        Core.farmStatusString = string.format("[ACTIVE - COLLECTING %d/%d]", Core.bagCoinsActual, Core.bagCoinsMax)
                        Core.farmEnSpawn = false
                        Core.farmIsUnderground = true
                        Core.farmActivarNoclip()
                        Core.farmActivarXRay()
                        Core.farmActivarFisicas(hrp, humanoid)
                        Core.farmActivarCamara(hrp)

                        local posBajoTierra = moneda.Position - Vector3.new(0, Core.PROFUNDIDAD_SUELO, 0)
                        local cfObjetivo = CFrame.new(posBajoTierra) * CFrame.Angles(math.rad(-90), 0, 0)
                        local duracion = math.clamp((hrp.Position - posBajoTierra).Magnitude / Core.farmSpeedVal, 0.05, 3.0)

                        local tween = TweenService:Create(hrp, TweenInfo.new(duracion, Enum.EasingStyle.Linear), {CFrame = cfObjetivo})
                        tween:Play()
                        tween.Completed:Wait()

                        if not Core.autoFarmEnabled then break end
                        Core.FarmInternal.ignoredCoins[moneda] = os.clock() + 6

                        if moneda.Parent then
                            pcall(function()
                                if typeof(firetouchinterest) == "function" then
                                    firetouchinterest(hrp, moneda, 0)
                                    task.wait()
                                    firetouchinterest(hrp, moneda, 1)
                                else
                                    local oldCF = hrp.CFrame
                                    hrp.CFrame = moneda.CFrame
                                    task.wait(0.02)
                                    hrp.CFrame = oldCF
                                end
                            end)
                        end
                        task.wait(Core.TIEMPO_ESPERA)
                    else
                        Core.farmStatusString = "[STANDBY - WAITING NEXT ROUND]"
                        if Core.farmIsUnderground then Core.farmVolverASuperficie(hrp) end
                        Core.farmDesactivarNoclip()
                        Core.farmDesactivarXRay()
                        Core.farmRestaurarCamara()
                        Core.farmRestaurarFisicas(humanoid)
                        task.wait(0.8)
                    end
                end
            end
        else
            task.wait(0.5)
        end
    end
end

                end
            end

                elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
                    if Core.fpsRestoreCache.particles[v] == nil then Core.fpsRestoreCache.particles[v] = v.Enabled end
                    v.Enabled = false
                end
            end
            if Core.Notify then Core.Notify("Performance", "Modo Ultra FPS ACTIVADO.", 3) end
        else
            Lighting.GlobalShadows = Core.fpsRestoreCache.globalShadows
            Lighting.FogEnd = Core.fpsRestoreCache.fogEnd
            if terrain then
                terrain.WaterWaveSize = Core.fpsRestoreCache.waterWaveSize
                terrain.WaterReflectance = Core.fpsRestoreCache.waterReflectance
            end
            for eff, wasEnabled in pairs(Core.fpsRestoreCache.effects) do if eff and eff.Parent then eff.Enabled = wasEnabled end end
            for part, data in pairs(Core.fpsRestoreCache.parts) do if part and part.Parent then part.Material = data.Material part.Reflectance = data.Reflectance end end

            table.clear(Core.fpsRestoreCache.decals)
            table.clear(Core.fpsRestoreCache.particles)
            table.clear(Core.fpsRestoreCache.effects)
            if Core.Notify then Core.Notify("Performance", "Gráficos originales RESTAURADOS.", 3) end
        end
    end)

Core.killAuraActive = false
Core.killAuraRange = 16
Core.killAuraDebounce = false

task.spawn(function()
    while Core.scriptActive do
        if Core.killAuraActive and not Core.killAuraDebounce and Core.isPlayerAlive and Core.isPlayerAlive(LocalPlayer) then
            pcall(function()
                local char = LocalPlayer.Character
                local backpack = LocalPlayer:FindFirstChild("Backpack")
                local knife = char and (char:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife")))

                if knife then
                    local myHRP = char:FindFirstChild("HumanoidRootPart")
                    local hum = char:FindFirstChildOfClass("Humanoid")

                    if myHRP and hum and hum.Health > 0 then
                        local effectiveRange = math.clamp(Core.killAuraRange or 16, 5, 30)

                        for _, player in ipairs(Players:GetPlayers()) do
                            if player ~= LocalPlayer and Core.isPlayerAlive(player) then
                                local tChar = player.Character
                                local tHRP = tChar:FindFirstChild("HumanoidRootPart")
                                local tHum = tChar:FindFirstChildOfClass("Humanoid")

                                if tHRP and tHum and tHum.Health > 0 then
                                    local dist = (myHRP.Position - tHRP.Position).Magnitude
                                    if dist <= effectiveRange then
                                        Core.killAuraDebounce = true

                                        if knife.Parent == backpack then
                                            hum:EquipTool(knife)
                                            task.wait(0.02)
                                        end

                                        if knife:FindFirstChild("Stab") then
                                            knife.Stab:FireServer("Slash")
                                        end

                                        local handle = knife:FindFirstChild("Handle")
                                        if handle and typeof(firetouchinterest) == "function" then
                                            pcall(function()
                                                firetouchinterest(tHRP, handle, 0)
                                                firetouchinterest(tHRP, handle, 1)
                                                if tChar:FindFirstChild("UpperTorso") then
                                                    firetouchinterest(tChar.UpperTorso, handle, 0)
                                                    firetouchinterest(tChar.UpperTorso, handle, 1)
                                                elseif tChar:FindFirstChild("Torso") then
                                                    firetouchinterest(tChar.Torso, handle, 0)
                                                    firetouchinterest(tChar.Torso, handle, 1)
                                                end
                                            end)
                                        end

                                        task.wait(0.12)
                                        Core.killAuraDebounce = false
                                        break
                                    end
                                end
                            end
                        end
                    end
                end
            end)
        end
        task.wait(0.08)
    end
end)

local function ApplyDrag(handle, frame)
    local isDragging = false
    local dragStart = Vector2.zero
    local startPos = UDim2.new()

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            isDragging = true
            dragStart = Vector2.new(input.Position.X, input.Position.Y)
            startPos = frame.Position

            local endConn
            endConn = UserInputService.InputEnded:Connect(function(endInput)
                if endInput.UserInputType == Enum.UserInputType.MouseButton1 or endInput.UserInputType == Enum.UserInputType.Touch then
                    isDragging = false
                    endConn:Disconnect()
                end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = Vector2.new(input.Position.X, input.Position.Y) - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function ToggleHub(open)
    IS_OPEN = open
    if open then
        TweenService:Create(FloatScale, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Scale = 0}):Play()
        task.delay(0.2, function() FloatingIcon.Visible = false end)
        MainHub.Visible = true
        TweenService:Create(HubScale, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
    else
        local shrink = TweenService:Create(HubScale, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.In), {Scale = 0})
        shrink:Play()
        shrink.Completed:Connect(function()
            MainHub.Visible = false
            FloatingIcon.Visible = true
            TweenService:Create(FloatScale, TweenInfo.new(0.35, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {Scale = 1}):Play()
        end)
    end
end

local isFloatDrag = false
local floatStartMouse = Vector2.zero
local floatStartPos = UDim2.new()

FloatingIcon.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isFloatDrag = true
        floatStartMouse = Vector2.new(input.Position.X, input.Position.Y)
        floatStartPos = FloatingIcon.Position

        local releaseConn
        releaseConn = UserInputService.InputEnded:Connect(function(endInput)
            if endInput.UserInputType == Enum.UserInputType.MouseButton1 or endInput.UserInputType == Enum.UserInputType.Touch then
                isFloatDrag = false
                releaseConn:Disconnect()

                local endMouse = Vector2.new(endInput.Position.X, endInput.Position.Y)
                if (endMouse - floatStartMouse).Magnitude < 10 then
                    ToggleHub(true)
                else
                    local vSize = Camera.ViewportSize
                    local curX = FloatingIcon.Position.X.Offset + (FloatingIcon.Position.X.Scale * vSize.X)
                    local curY = FloatingIcon.Position.Y.Offset + (FloatingIcon.Position.Y.Scale * vSize.Y)
                    local snapX = (curX < vSize.X * 0.5) and 35 or (vSize.X - 35)
                    local snapY = math.clamp(curY, 40, vSize.Y - 40)

                    TweenService:Create(FloatingIcon, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                        Position = UDim2.fromOffset(snapX, snapY)
                    }):Play()
                end
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isFloatDrag and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = Vector2.new(input.Position.X, input.Position.Y) - floatStartMouse
        FloatingIcon.Position = UDim2.new(floatStartPos.X.Scale, floatStartPos.X.Offset + delta.X, floatStartPos.Y.Scale, floatStartPos.Y.Offset + delta.Y)
    end
end)

    ValLabel.Font = Enum.Font.GothamBold
    ValLabel.TextSize = 10
    ValLabel.TextXAlignment = Enum.TextXAlignment.Right
    ValLabel.Parent = Card

    applyTextLaser(ValLabel, true)

    local Track = Instance.new("Frame")
    Track.Size = UDim2.new(1, -20, 0, 4)
    Track.Position = UDim2.new(0, 10, 0, 24)
    Track.BackgroundColor3 = Color3.fromRGB(30, 32, 40)
    Track.BorderSizePixel = 0
    Track.Parent = Card

    local TrackCorner = Instance.new("UICorner")
    TrackCorner.CornerRadius = UDim.new(1, 0)
    TrackCorner.Parent = Track

    local Fill = Instance.new("Frame")
    local pct = math.clamp((currentVal - min) / (max - min), 0, 1)
    Fill.Size = UDim2.new(pct, 0, 1, 0)
    Fill.BackgroundColor3 = THEME.Crimson
    Fill.BorderSizePixel = 0
    Fill.Parent = Track

    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(1, 0)
    FillCorner.Parent = Fill

    local dragging = false
    local function UpdateSlider(xPos)
        local rel = math.clamp((xPos - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
        currentVal = math.floor(min + ((max - min) * rel))
        ValLabel.Text = tostring(currentVal)
        Fill.Size = UDim2.new(rel, 0, 1, 0)
        if callback then callback(currentVal) end
    end

    Card.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            UpdateSlider(input.Position.X)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            UpdateSlider(input.Position.X)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end
